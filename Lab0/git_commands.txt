git init - создать новый репозиторий 
git clone <url> - скопировать удалённый репозиторий 
git status - показать состояние файлов 
git add <file> - добавить файл в список на коммит 
git commit -m "сообщение" - сохранить изменения 
git push - отправить изменения на сервер 
git pull - скачать обновления с сервера 
git branch - посмотреть ветки 
git checkout <branch> - переключиться на другую ветку 
git merge <branch> - слить ветки 
git log - показать историю коммитов
