git init – создание нового репозитория
git clone <url> – клонировать удалённый репозиторий
git status – показать состояние файлов
git add <file> – добавить файл в индекс
git commit -m "сообщение" – сохранить изменения с комментарием
git push – отправить изменения на сервер
git pull – получить обновления с сервера
git branch – список веток
git checkout <branch> – переключиться на ветку
git merge <branch> – слить ветки
git log – показать историю коммитов
